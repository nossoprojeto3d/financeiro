# Fiados — Plano de implementação

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Anotar quem levou e ainda não pagou e, no "Recebeu", virar uma Venda no caixa com a forma de pagamento escolhida.

**Architecture:** Tabela própria `fiados` no Supabase, mais a função `receber_fiado` que apaga o fiado e cria o lançamento na mesma transação. No app, os fiados entram no estado (`S.fiados`) com carregamento, sincronização e cache, iguais aos recorrentes. O formulário `Lancar.svelte` ganha um "modo fiado", e o componente novo `Fiados.svelte` mostra a linha do Início, o painel "A receber" e o painel "Recebeu".

**Tech Stack:** Svelte 5 (runes), Tailwind 4, Lucide, Supabase REST via `fetch`, PL/pgSQL, Vitest (novo, só para desenvolvimento).

**Spec:** `docs/superpowers/specs/2026-10-14-fiados-design.md`

## Global Constraints

- Tudo em português do Brasil: interface, nomes de funções e variáveis, comentários e commits (no imperativo, descrevendo o efeito).
- Dinheiro sempre em centavos: `cent()` e `soma()` do `app/src/lib/formato.js`. Nunca somar decimais direto.
- Campos de texto com `text-[16px]` ou maior e alvos de toque com pelo menos 40px (`h-10`).
- Ações que gravam começam com `if (enviando) return`.
- Não usar `confirm()`, `alert()` nem `prompt()`.
- Não renomear chaves do `localStorage` (`caixa_*`). Os fiados entram como campo novo dentro do `caixa_cache`.
- Linguagem para leigos e tela limpa: nada de texto extra além do que está na spec.
- Não editar arquivos gerados na raiz (`index.html`, `assets/`, `sw.js`...). Editar em `app/` e rodar o build.
- **Nunca tocar no banco de verdade nos testes.** O SQL quem roda é o Junior. Nada de push sem autorização.
- Trabalhar no branch `fiados` (push na `main` publica direto).

## Review Focus

1. **Toque duplo ou os dois celulares no "Recebeu":** só pode existir uma Venda. O segundo pedido recebe "Esse fiado já foi recebido ou excluído." e a lista recarrega. Teste: Task 6, passo 4 (cenário E).
2. **Queda de internet só na busca de fiados durante a sincronização:** a lista não pode sumir. Só a resposta "tabela não encontrada" esconde os fiados. Teste: Task 6, passo 4 (cenário H).
3. **Rascunho de fiado com só o nome preenchido:** a sincronização de 30 s não pode trocar o formulário, e trocar de tela não pode descartar o rascunho. Teste: Task 6, passo 4 (cenário C).
4. **Excluir categoria que só tem fiados:** o app pede outra categoria e move os fiados, em vez de dar erro de chave estrangeira. Teste: Task 6, passo 4 (cenário I).
5. **Data do pagamento antes da entrega:** o seletor não deixa escolher e o SQL recusa. Teste: Task 1, passo 2 (revisão do SQL) e Task 6, passo 4 (cenário D).

```sql
--  Financeiro NP3D — fiados (versão 4.1)
--  Quem levou e ainda não pagou. Rode este arquivo inteiro no SQL Editor
--  depois dos outros (setup, v2, v2-ordem, v3-seguranca).
--  Pode rodar mais de uma vez sem problema.

create table if not exists public.fiados (
  id bigint generated always as identity primary key,
  nome text not null check (char_length(btrim(nome)) between 1 and 60),
  valor numeric(12,2) not null check (valor > 0),
  categoria_id bigint not null references public.categorias(id),
  descricao text not null check (char_length(btrim(descricao)) between 1 and 200),
  data date not null default current_date
    check (data between date '2000-01-01' and (current_date + 1)),
  usuario_id uuid not null default auth.uid() references public.perfis(id),
  criado_em timestamptz not null default now(),
  atualizado_em timestamptz not null default now()
);

drop trigger if exists fiados_atualizado on public.fiados;
create trigger fiados_atualizado before update on public.fiados
  for each row execute function public.tocar_atualizado();

alter table public.fiados enable row level security;
drop policy if exists fiados_membros on public.fiados;
create policy fiados_membros on public.fiados for all to authenticated
  using ((select public.is_membro())) with check ((select public.is_membro()));
revoke all on public.fiados from anon;
grant select, insert, update, delete on public.fiados to authenticated;

create index if not exists fiados_categoria_idx on public.fiados (categoria_id);

-- Recebeu: apaga o fiado e lança a venda juntos. Se dois pedidos chegarem
-- ao mesmo tempo (toque duplo, os dois celulares), o segundo não acha mais
-- o fiado e não lança nada.
create or replace function public.receber_fiado(p_id bigint, p_forma text, p_data date)
returns public.lancamentos language plpgsql security definer set search_path = '' as $$
declare
  f public.fiados;
  novo public.lancamentos;
  hoje date := (now() at time zone 'America/Sao_Paulo')::date;
begin
  if not public.is_membro() then raise exception 'sem permissão'; end if;
  if char_length(coalesce(btrim(p_forma), '')) not between 1 and 40 then
    raise exception 'Escolha a forma de pagamento.';
  end if;
  delete from public.fiados where id = p_id returning * into f;
  if not found then raise exception 'Esse fiado já foi recebido ou excluído.'; end if;
  if p_data is null or p_data > hoje or p_data < f.data then
    raise exception 'A data do pagamento não pode ser antes da entrega nem no futuro.';
  end if;
  insert into public.lancamentos (tipo, valor, categoria_id, descricao, forma_pagamento, data, usuario_id)
  values ('entrada', f.valor, f.categoria_id, left(btrim(f.nome) || ' · ' || btrim(f.descricao), 200),
          btrim(p_forma), p_data, auth.uid())
  returning * into novo;
  return novo;
end $$;

revoke all on function public.receber_fiado(bigint, text, date) from public, anon;
grant execute on function public.receber_fiado(bigint, text, date) to authenticated;

-- Conferência: deve mostrar a tabela e a função
select 'fiados' as criado, to_regclass('public.fiados') is not null as ok
union all
select 'receber_fiado', exists (select 1 from pg_proc where proname = 'receber_fiado');
```

```js
// Fiados: quem levou e ainda não pagou. Contas puras (sem banco nem tela).
import { deISO } from './formato.js'

// O mesmo nome escrito de jeitos diferentes ("Ana", " ana ", "Ána") é uma pessoa só
const chave = nome => String(nome || '').trim().toLocaleLowerCase('pt-BR').normalize('NFD').replace(/[̀-ͯ]/g, '')

export const pessoas = lista => new Set(lista.map(f => chave(f.nome))).size

// Sugestões do campo "Quem deve": um de cada, com a primeira grafia usada
export function nomesUsados(lista) {
  const vistos = new Map()
  for (const f of lista) {
    const k = chave(f.nome)
    if (k && !vistos.has(k)) vistos.set(k, f.nome.trim())
  }
  return [...vistos.values()].sort((a, b) => a.localeCompare(b, 'pt-BR'))
}

export function haQuanto(dataISO, hojeISO) {
  const dias = Math.round((deISO(hojeISO) - deISO(dataISO)) / 864e5)
  if (dias <= 0) return 'hoje'
  if (dias === 1) return 'ontem'
  return `há ${dias} dias`
}

// Atalho "Anotar fiado": já vem marcada a primeira categoria de venda
export const categoriaPadraoFiado = cats => cats.find(c => c.natureza === 'venda') ?? cats[0] ?? null

// Quem deve há mais tempo aparece primeiro
export const ordenarFiados = lista => lista.sort((a, b) => (a.data < b.data ? -1 : a.data > b.data ? 1 : a.id - b.id))
```

```js
/* ---------- Fiados ---------- */
const SUMIU = 'Esse fiado já foi recebido ou excluído.'

async function salvarFiado(f) {
  const dados = { nome: f.nome.trim(), valor: f.centavos / 100, categoria_id: f.categoria_id,
    descricao: f.descricao.trim(), data: f.data, usuario_id: f.usuario_id }
  if (f.id) {
    const r = await Api.editarFiado(f.id, dados)
    if (!r) {
      S.fiados = S.fiados.filter(x => x.id !== f.id)
      guardarCache()
      throw new Error(SUMIU)
    }
    const i = S.fiados.findIndex(x => x.id === f.id)
    if (i >= 0) S.fiados[i] = numerico(r); else S.fiados.push(numerico(r))
  } else {
    S.fiados.push(numerico(await Api.criarFiado(dados)))
  }
  ordenarFiados(S.fiados)
  guardarCache()
}

export async function excluirFiado(id) {
```

```js
    if (caminho === 'fiados' && db.semTabela) return resp({ code: 'PGRST205', message: "Could not find the table 'public.fiados' in the schema cache" }, 404)
    if (caminho === 'fiados' && m === 'GET' && db.redeCaiNosFiados) throw new TypeError('Load failed')
    const lista = db[caminho]
    if (!lista) return resp({ message: 'tabela desconhecida: ' + caminho }, 404)
    const filtros = [...u.searchParams].filter(([k, v]) => v.startsWith('eq.'))
    const bate = x => filtros.every(([k, v]) => String(x[k]) === v.slice(3))
    if (m === 'GET') return resp(lista)
    if (m === 'POST') { const novo = { ...corpo, id: ++seq, atualizado_em: 't' + seq }; lista.push(novo); return resp([novo], 201) }
    if (m === 'PATCH') { const alvo = lista.filter(bate); alvo.forEach(x => Object.assign(x, corpo, { atualizado_em: 't' + ++seq })); return resp(alvo) }
    if (m === 'DELETE') { for (let i = lista.length - 1; i >= 0; i--) if (bate(lista[i])) lista.splice(i, 1); return resp(null, 204) }
    return resp({ message: 'método?' }, 405)
  }
})()
```
